Only rendered for posts that opted into a contents block and that have enough sections for the rail to tell you anything. */ -}} {{- $sections := slice -}} {{- range .Fragments.Headings -}} {{- range .Headings }}{{ $sections = $sections | append . }}{{ end -}} {{- end -}} {{- if and .Params.toc (ge (len $sections) 3) -}}
