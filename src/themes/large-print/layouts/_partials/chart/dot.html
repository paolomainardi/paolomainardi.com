{{- /* Dot plot. One row per category, one mark per series on a shared scale. Takes a dict with "rows" (from chart/data.html) and "p", the chart params. In small multiples it also takes "all", every row of the chart, which sets the scale and the series so the panels match, and "key" false, because the key is drawn once above the panels. The params: y column naming the row category x column holding the value series optional column naming the series; each gets its own shape lo, hi optional columns with the low and high of each value, drawn as a whisker through the mark highlight optional series drawn in the accent, with its values labelled; without series, the row category drawn that way label optional comma separated series (or row categories) whose values are labelled too, without the accent unit optional unit for the last tick and the hover titles from "0" (default) starts the scale at zero, "auto" at the data Marks are positioned in percent of the track, so the chart reflows with the column and its text stays at reading size on a phone. */ -}} {{- $p := .p -}} {{- $rows := .rows -}} {{- $unit := cond (ne $p.unit "") (printf " %s" $p.unit) "" -}} {{- $all := .all | default $rows -}} {{- /* Categories in order of first appearance, and the extent. */ -}} {{- $cats := slice -}} {{- range $rows -}} {{- $c := index . $p.y -}} {{- if not (in $cats $c) }}{{ $cats = $cats | append $c }}{{ end -}} {{- end -}} {{- $lo := 0.0 -}} {{- $hi := 0.0 -}} {{- range $i, $r := $all -}} {{- $a := float (index $r (or $p.lo $p.x)) -}} {{- $b := float (index $r (or $p.hi $p.x)) -}} {{- if or (eq $i 0) (lt $a $lo) }}{{ $lo = $a }}{{ end -}} {{- if or (eq $i 0) (gt $b $hi) }}{{ $hi = $b }}{{ end -}} {{- end -}} {{- if and (ne $p.from "auto") (gt $lo 0.0) }}{{ $lo = 0.0 }}{{ end -}} {{- $scale := partial "chart/ticks.html" (dict "lo" $lo "hi" $hi) -}} {{- $min := $scale.min -}} {{- $span := sub $scale.max $scale.min -}} {{- $shapes := partial "chart/shapes.html" (dict "rows" $all "p" $p) -}} {{- $shapeOf := $shapes.shapeOf -}} {{- if ne .key false }} {{- partial "chart/key.html" (dict "shapes" $shapes "highlight" $p.highlight) }} {{- end }}
{{- range $cats }} {{- $cat := . }} {{- $mine := where $rows $p.y $cat }} {{- /* rlo and rhi span the values in the row, rend is the right end of everything drawn in it, where a value label can sit clear of the marks. */}} {{- $rlo := 0.0 }} {{- $rhi := 0.0 }} {{- $rend := 0.0 }} {{- range $i, $r := $mine }} {{- $v := float (index $r $p.x) }} {{- $e := float (index $r (or $p.hi $p.x)) }} {{- if or (eq $i 0) (lt $v $rlo) }}{{ $rlo = $v }}{{ end }} {{- if or (eq $i 0) (gt $v $rhi) }}{{ $rhi = $v }}{{ end }} {{- if gt $e $rend }}{{ $rend = $e }}{{ end }} {{- end }} {{- $labelAt := mul (div (sub $rend $min) $span) 100 }}
{{ $cat }}{{ with index $p.notes $cat }}{{ . }}{{ end }}
