{{- /* Dot plot. One row per category, one mark per series on a shared scale. Takes a dict with "rows" (from chart/data.html) and "p", the chart params: y column naming the row category x column holding the value series optional column naming the series; each gets its own shape lo, hi optional columns with the low and high of each value, drawn as a whisker through the mark highlight optional series drawn in the accent, with its values labelled unit optional unit for the last tick and the hover titles from "0" (default) starts the scale at zero, "auto" at the data Marks are positioned in percent of the track, so the chart reflows with the column and its text stays at reading size on a phone. */ -}} {{- $p := .p -}} {{- $rows := .rows -}} {{- $unit := cond (ne $p.unit "") (printf " %s" $p.unit) "" -}} {{- /* Categories and series in order of first appearance, and the extent. */ -}} {{- $cats := slice -}} {{- $series := slice -}} {{- $lo := 0.0 -}} {{- $hi := 0.0 -}} {{- range $i, $r := $rows -}} {{- $c := index $r $p.y -}} {{- if not (in $cats $c) }}{{ $cats = $cats | append $c }}{{ end -}} {{- if $p.series -}} {{- $s := index $r $p.series -}} {{- if not (in $series $s) }}{{ $series = $series | append $s }}{{ end -}} {{- end -}} {{- $a := float (index $r (or $p.lo $p.x)) -}} {{- $b := float (index $r (or $p.hi $p.x)) -}} {{- if or (eq $i 0) (lt $a $lo) }}{{ $lo = $a }}{{ end -}} {{- if or (eq $i 0) (gt $b $hi) }}{{ $hi = $b }}{{ end -}} {{- end -}} {{- if and (ne $p.from "auto") (gt $lo 0.0) }}{{ $lo = 0.0 }}{{ end -}} {{- $scale := partial "chart/ticks.html" (dict "lo" $lo "hi" $hi) -}} {{- $min := $scale.min -}} {{- $span := sub $scale.max $scale.min -}} {{- /* The highlighted series is a circle in the accent; the others take a shape each, in ink, so they stay distinguishable without colour. */ -}} {{- $shapes := slice "circle" "square" "diamond" "triangle" -}} {{- $shapeOf := dict -}} {{- $n := 0 -}} {{- range $series -}} {{- if eq . $p.highlight -}} {{- $shapeOf = merge $shapeOf (dict . "circle hot") -}} {{- else -}} {{- $shapeOf = merge $shapeOf (dict . (index $shapes (mod $n (len $shapes)))) -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- if gt (len $series) 1 }}
