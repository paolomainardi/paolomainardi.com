{{- /* Bar chart, horizontal. One bar per row, always from zero, with its value written at the end of the bar. Takes a dict with "rows" (from chart/data.html) and "p", the chart params. In small multiples it also takes "all", every row of the chart, which sets the scale so the panels match. The params: y column naming the bar x column holding the value highlight optional bar name drawn in the accent unit optional unit for the last tick and the hover titles */ -}} {{- $p := .p -}} {{- $rows := .rows -}} {{- $unit := cond (ne $p.unit "") (printf " %s" $p.unit) "" -}} {{- $hi := 0.0 -}} {{- range .all | default $rows }}{{ $v := float (index . $p.x) }}{{ if gt $v $hi }}{{ $hi = $v }}{{ end }}{{ end -}} {{- $scale := partial "chart/ticks.html" (dict "lo" 0 "hi" $hi) -}} {{- $span := sub $scale.max $scale.min }}
{{- range $rows }} {{- $name := index . $p.y }} {{- $v := index . $p.x }} {{- $w := mul (div (float $v) $span) 100 }}
{{ $name }}{{ with index $p.notes $name }}{{ . }}{{ end }}
{{ $v }}
{{- end }} {{- partial "chart/axis.html" (dict "scale" $scale "unit" $p.unit "title" $p.axis) }}
